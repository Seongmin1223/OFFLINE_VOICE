$OUTPUT_BASE = "D:\shin\종프2"
$LLAMA_CPP = "D:\shin\llama.cpp"
$GGUF_DIR = "$OUTPUT_BASE\gguf"

New-Item -ItemType Directory -Force -Path $GGUF_DIR | Out-Null

foreach ($rank in @(8, 16, 32)) {
    $merged_path = "$OUTPUT_BASE\poby_merged_r$rank"
    $fp16_gguf   = "$GGUF_DIR\poby_r${rank}_fp16.gguf"

    Write-Host "=== Rank $rank FP16 GGUF 변환 중 ===" -ForegroundColor Cyan

    python "$LLAMA_CPP\convert_hf_to_gguf.py" $merged_path --outfile $fp16_gguf --outtype f16

    Write-Host "Q4_K_M 양자화 중..." -ForegroundColor Yellow
    & "$LLAMA_CPP\build\bin\Release\llama-quantize.exe" $fp16_gguf "$GGUF_DIR\poby_r${rank}_q4km.gguf" Q4_K_M

    Write-Host "Q5_K_M 양자화 중..." -ForegroundColor Yellow
    & "$LLAMA_CPP\build\bin\Release\llama-quantize.exe" $fp16_gguf "$GGUF_DIR\poby_r${rank}_q5km.gguf" Q5_K_M

    Write-Host "Q8_0 양자화 중..." -ForegroundColor Yellow
    & "$LLAMA_CPP\build\bin\Release\llama-quantize.exe" $fp16_gguf "$GGUF_DIR\poby_r${rank}_q8.gguf" Q8_0

    Write-Host "Rank $rank 완료!" -ForegroundColor Green
}

Write-Host "모든 변환 완료!" -ForegroundColor Green
